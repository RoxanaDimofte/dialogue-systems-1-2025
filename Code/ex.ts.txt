function makeGuess(answers) {
    for (let i = 0; i < people.length; i++) {
        let match = true;

        for (const key in answers) {
            if (answers[key] !== people[i][key]) {
                match = false;
                break;
            }
        }

        if (match) {
            return people[i].name;
        }
    }
    return null;
}
